Boss::Boss() : hp(0.0f), state(BossState::WAITING) {
	for (int i = 0; i < 20; i++) {
		BossGraphs[i] = -1;
	}

	BossGraphs[1] = LoadGraph("素材集/改良版虹色大玉.png");
}

void Boss::Spawn(float in_x, float in_y, const BossPhase* phaseTable, int phaseCount, int startPhase, int in_endIdx) {
	if (phaseTable == nullptr || phaseCount <= 0) return;		//テーブルが不正な場合の早期return
	
	BulletPattern::ShotConfig dummyConf;
	Encount(in_x, in_y, para.r, 0.0f, 0.0f, 0, dummyConf, false);							//hpの引数はボスの場合hpロジックが独自の為ダミーの0を入力している。その後上書きされる為何でも良い。

	this->m_phases = phaseTable;																//渡された弾幕テーブルのポインタを保持
	this->m_phaseMax = phaseCount;																//渡された弾幕テーブルの要素数を保持
	this->currentIdx = (startPhase < phaseCount) ? startPhase : 0;								//startPhaseが要素数を超えないようにクランプ
	this->endIdx = (in_endIdx >= 0 && in_endIdx < phaseCount) ? in_endIdx : (phaseCount - 1);	//終了フェーズを設定(-1が渡された場合は最終フェーズまで)

	this->phaseMaxHp = m_phases[currentIdx].hpLimit;					//フェーズの最大HPをセット
	this->hp = 0.0f;													//初期スポーン時のhpチャージ演出の起点とするため,hp=0.0fとしておく

	this->timer = 0;
	this->phaseTimer = 0;
	this->flag = true;
	this->state = BossState::WAITING;									//スポーン時は待機状態として初期化
	this->waitTimer = 0;
	this->chargeTimer = 0;

}


void Boss::Update(const Player& player, BulletManager* bm) {									//BulletManagerを使って弾を撃つ
	using namespace PlayArea;
	if (!flag || m_phases == nullptr) return;	//ボスフラグが無い、又はm_phasesがヌルポインタの場合は早期return

	if (state == BossState::ACTIVE) {		//戦闘状態
		timer++;
		phaseTimer++;


		x = PlayArea::DefaultBossX;
		y = PlayArea::FitfullBossY;
		const BossPhase& p = m_phases[currentIdx];

		for (int i = 0; i < p.shotCount; i++) {
			const auto& conf = p.shots[i];

			if (phaseTimer % conf.interval == 0) {
				if (conf.type == PT::RotateAll) {
					float baseAngleDeg = conf.baseAngle + (timer * conf.rotSpeed) * 180.0f / DX_PI_F;
					BulletPattern::NWayShot(x, y, conf.radius, conf.speed, conf.way, conf.totalAngle, baseAngleDeg, bm, conf.graphicType);
				}
				else {
					BulletPattern::ExecShot(x, y, conf, player, bm);
				}
			}
		}

		int maxFrames = (int)(p.limitTimeSeconds * FPSManager::GetTargetFPS());		//maxFrames[F]=制限時間[s]*FPS[F/s]

		if (hp <= 0 || phaseTimer >= maxFrames) {							//フェーズのhpがなくなるか、limitTimer(制限時間)を越えた場合

			for (int typeIdx = 0; typeIdx < DROP_TYPE_MAX; typeIdx++) {					//アイテム全種類のループ
				int count = m_phases[currentIdx].dropCounts[typeIdx];				//静的配列から値を取得して格納			
				if (count > 0) {												//1以上の時
					ItemType iType = static_cast<ItemType>(typeIdx);		//アイテムの種類を格納
					bm->LaunchItem(x, y, iType, count, 35.0f);				//ボス座標から半径35.0f以内にランダムに散布してドロップ
				}
			}

			if (m_phases[currentIdx].clearOnEnd) {
				bm->ConvertEnemyBulletsToItems();												//現在のphaseでclearOnEndフラグが立っているならbmに弾消し申請
			}

			currentIdx++;															//現在のフェーズを次の段階へ
			if (currentIdx <=endIdx && currentIdx<m_phaseMax) {							//endIdx以下且つ、フェーズが最大まで終わっていない場合
				state = BossState::WAITING;									//待機状態遷移
				waitTimer = 0;												//待機時間タイマを初期化

				this->phaseMaxHp = m_phases[currentIdx].hpLimit;				//フェーズの最大hp変数にそのフェーズの最大hpを格納
				this->hp = 0.0f;											//hpチャージ演出の起点とするため,hp=0.0fとしておく
			}
			else {
				bm->ConvertEnemyBulletsToItems();									//全スペカ終了で弾全消し
				flag = false;														//全スペカ終了でボスフラグを消す
			}
		}
	}
	else if (state == BossState::WAITING) {		//待機状態
		waitTimer++;															//待機時間タイマのカウントアップ開始

		this->hp = 0;

		if (waitTimer >= PHASE_WAIT_FRAME) {	//待機時間タイマのカウントがフェーズの待機フレーム以上に成った場合
			state = BossState::CHARGING;						//待機状態が終わったらチャージ状態へ遷移
			chargeTimer = 0;								//チャージタイマをリセット
		}
	}
	else if (state == BossState::CHARGING) {
		chargeTimer++;
		this->hp= this->phaseMaxHp * ((float)chargeTimer / PHASE_CHARGE_FRAME);	//hpチャージ演出(0からphaseMaxHpまで、PHASE_CHARGE_FRAMEをかけて線形補間で増加させる)

		if (chargeTimer >= PHASE_CHARGE_FRAME) {
			this->hp = this->phaseMaxHp;										//hpに確実に最大値を格納
			this->phaseTimer = 0;												//フェーズタイマーの初期化
			state = BossState::ACTIVE;											//戦闘状態へ遷移
		}
	}
}


void Boss::Draw() {
	if (!flag)return;
	float scale = r / 128.0f;


	float angle = atan2f(vy, vx);



	if (BossGraphs[1] != -1) {
		DrawRotaGraph((int)x, (int)y, scale, angle, BossGraphs[1], TRUE);
	}
	else {
		DrawCircle(x, y, r, GetColor(255, 0, 0), TRUE);
	}

	if ((state == BossState::ACTIVE || state == BossState::CHARGING) && phaseMaxHp > 0) {			//戦闘状態か(チャージ状態且つ体力が0以上)の場合
		for (int i = 0; i < para.GaugeSegments; i++) {									//多角形の角数だけ繰り返す
			float a1 = para.StartAngle - (DX_PI_F * 2.0f * i / para.GaugeSegments);			//現在の点の角度
			float a2 = para.StartAngle - (DX_PI_F * 2.0f * (i + 1) / para.GaugeSegments);	//次点の角度

			int x1 = (int)(x + cos(a1) * (r + para.GaugeOffset_R));								//現在の点のx座標
			int y1 = (int)(y + sin(a1) * (r + para.GaugeOffset_R));								//現在の点のy座標
			int x2 = (int)(x + cos(a2) * (r + para.GaugeOffset_R));								//次点のx
			int y2 = (int)(y + sin(a2) * (r + para.GaugeOffset_R));								//次点のy

			DrawLine(x1, y1, x2, y2, GetColor(64, 64, 64), para.GaugeThickness);				//グレーの体力ゲージ(体力に依存しない)の描画
		}

		float hp_rate = GetHpRate();										//hpの比率計算	
		int draw_limit = (int)(para.GaugeSegments * hp_rate);					//残HP分の角を描画

		for (int i = 0; i < draw_limit; i++) {									//残HP分の多角形の角数だけ繰り返す
			float a1 = para.StartAngle - (DX_PI_F * 2.0f * i / para.GaugeSegments);			//現在の点の角度
			float a2 = para.StartAngle - (DX_PI_F * 2.0f * (i + 1) / para.GaugeSegments);	//次点の角度

			int x1 = (int)(x + cos(a1) * (r + para.GaugeOffset_R));								//現在の点のx座標
			int y1 = (int)(y + sin(a1) * (r + para.GaugeOffset_R));								//現在の点のy座標
			int x2 = (int)(x + cos(a2) * (r + para.GaugeOffset_R));								//次点のx
			int y2 = (int)(y + sin(a2) * (r + para.GaugeOffset_R));								//次点のy


			DrawLine(x1, y1, x2, y2, GetColor(255, 255, 255), para.GaugeThickness);				//体力ゲージの描画
		}
	}

	if (state == BossState::ACTIVE) {			//ボスが戦闘状態中の場合
		int remainingFrames = GetRemainingPhaseTimer();			//そのフェーズの制限時間(整数)を格納
		int targetFPS = FPSManager::GetTargetFPS();				//1秒間のフレーム数(整数)を取得
		int seconds = remainingFrames / targetFPS;				//[s]=[F]/[F/s]
		
		if (seconds > 99) {
			seconds = 99;
		}

		unsigned int timeColor = ObjectColor::Get(ObjectColor::Type::WHITE);		//通常時の描画色
		if (seconds <= PhaseTimerColorChangeSeconds) timeColor = ObjectColor::Get(ObjectColor::Type::RED);		//5秒以下の描画色

		DrawFormatString(PhaseTimerX, PhaseTimerY, timeColor, "%02d", seconds);		//整数2桁で適当な色と位置で描画する(%02は2桁固定)
	}
}

bool Boss::CheckCollision(Player& player, BulletManager& bm) {
	if (!flag) return false;

	for (int i = 0; i < bm.GetPlayerBulletMax(); i++) {				//GetBulletNumは自機弾の最大数
		auto& b = bm.GetPlayerBullet(i);							// BulletManagerにこのゲッターが必要
		if (!b.flag) continue;										//その弾flagが立っていないなら以下の判定を飛ばす

		if (Collision::CheckCircle(x, y, r, b.x, b.y, b.r)) {		//ボスのx,y,rと自機弾のx,y,rを引数とする
			if (state == BossState::ACTIVE) {
				SetDamage(1.0f);										//1弾あたりのダメージ量
				
				player.AddScore(10);

			}
			b.flag = false;											//当たった弾フラグは消す
		}
	}

	if (state == BossState::ACTIVE && hp <= 0.0f) {				//hpが0以下になった時にtrueを返す
		return true;																			//倒したフラグを立てる
	}
	return false;
}

int Boss::GetRemainingPhaseTimer() const {		//外部から残り時間を取得するゲッタ関数
	if (state == BossState::WAITING || m_phases == nullptr) return 0;	//待機状態かm_phasesがヌルポインタの場合はの安全ガード

	int maxFrames = (int)(m_phases[currentIdx].limitTimeSeconds * FPSManager::GetTargetFPS());	//制限時間[F]=制限時間[s]*FPS[F/s]
	int remain = maxFrames - phaseTimer;														//そのフェーズの制限時間 - そのフェーズが開始してからの時間
	return (remain < 0) ? 0 : remain;															//制限時間よりも経過時間が大きくなったらremain<0より0を返し、そうでなければその差分の時間を返す
}		

void Boss::JumpToPhase(int phaseIdx) {
	if (m_phases == nullptr || phaseIdx < 0 || phaseIdx >= m_phaseMax) return;		//最大フェーズ配列数を越えない対策
	currentIdx = phaseIdx;
	phaseTimer = 0;
	phaseMaxHp = m_phases[currentIdx].hpLimit;
	hp = m_phaseMax;

	flag = true;
	state = BossState::WAITING;								//フェーズジャンプ時も待機状態からスタート
	waitTimer = 0;
}

struct BossParameter {
	float r;											//ボス半径
	float GaugeOffset_R;								//体力ゲージ半径
	int GaugeSegments;									//体力ゲージ円弧の分割数
	float GaugeThickness;								//体力ゲージの線の太さ
	float StartAngle;									//体力ゲージの開始角度
	float SpawnRate_X;									//初期X座標比率
	float SpawnRate_Y;									//初期Y座標比率
};

enum class BossState {
	ACTIVE,												//戦闘中
	WAITING,											//フェーズ間の待機中
	CHARGING											//チャージ演出中
};

class Boss : public Enemy {
private:
	
	static constexpr BossParameter para{
		30.0f,													//r：ボス半径
		25.0f,													//GaugeOffset_R：体力ゲージの半径		
		512,													//GaugeSegment：体力ゲージ円弧の分割数
		5.0f,													//GaugeThickness：体力ゲージの線の太さ
		-DX_PI_F / 2.0f,										//StartAngle：体力ゲージの開始角度。DX_PI_FはπでFはfloat型を示す。-π/2して真上を始点とする。
		0.5f,													//SpawnRate_X：プレイ領域X軸の左端を0、右端を1とした時の比率
		0.25f													//SpawnRate_Y：プレイ領域Y軸の上端を0、下端を1とした時の比率
	};

	static constexpr int PHASE_WAIT_FRAME = 120;				//フェーズ間の待機フレーム数
	static constexpr int PHASE_CHARGE_FRAME = 60;				//体力ゲージが溜まるチャージ演出にかかるフレーム数
	static constexpr int PhaseTimerColorChangeSeconds = 5;		//フェーズタイマの色の変化が起こる残り秒数

	const BossPhase* m_phases = nullptr;						//外部から渡されるフェーズ配列の戦闘ポインタを保持
	int m_phaseMax = 0;											//外部から渡されたフェーズ配列の要素数を保持

	int currentIdx = 0;											//現在のフェーズ番号
	int endIdx = 0;												//このボスの終了フェーズの番号
	
	int phaseTimer = 0;											//フェーズ開始からの経過時間
	float phaseMaxHp = 0.0f;									//現在のフェーズの初期HP
						
	BossState state;											//ボスの現在の状態を保持する変数
	int waitTimer = 0;											//待機時間をカウントするタイマ

	int chargeTimer = 0;										//チャージ演出の経過時間をカウントするタイマ
	int BossGraphs[20];											//画像番号を格納する箱

	float hp;											//現在の体力
	int timer = 0;

public:

	Boss();
	void ReloadGraphic() {
		BossGraphs[1] = LoadGraph("素材集/改良版虹色大玉.png");
	}

	void Spawn(float in_x, float in_y, const BossPhase* phaseTable, int phaseCount, int startPhase = 0, int in_endPhase = -1);				

	void Update(const class Player& player, class BulletManager* bm) override;				//BulletManagerを受け取ることで弾を出せるようにする
	void Draw() override;

	
	void SetDamage(float d) { hp -= d; }											
	float GetHp() { return hp; }						
	float GetHpRate() { return(phaseMaxHp > 0.0f) ? (hp / phaseMaxHp) : 0.0f; }		//そのフェーズの残りhpを円弧のUIで表現する時用の比率

	int GetRemainingPhaseTimer() const;												//外部(UI描画等)から現在のフェーズの制限時間の残りフレーム数を取得するためのゲッタ
	bool IsDead() const { return (m_phases != nullptr && !flag && currentIdx >= m_phaseMax); }	//ボスフラグが折れていてフェーズが最大(ラストスペルの次のフェーズ遷移)でクリア判定を返すゲッタ


	bool CheckCollision(class Player& player, class BulletManager& bm);				//冒頭でインクルードしなかったのは循環参照を防ぐため。それより、ここで前方宣言をする。

	void Reset() {
		flag = false;
		currentIdx = 0;
		m_phases = nullptr;		//ポインタ初期化
		m_phaseMax = 0;			//要素数初期化
	}

	void JumpToPhase(int phaseIdx);
};

void Bullet::Guide(float tx, float ty, float turnSpeed) {
	float targetAngle = atan2f(ty - y, tx - x);
	float currentAngle = atan2f(vy, vx);

	float radgap = targetAngle - currentAngle;
	while (radgap > DX_PI_F) radgap -= DX_PI_F * 2.0f;
	while (radgap < -DX_PI_F) radgap += DX_PI_F * 2.0f;

	float turn = radgap * turnSpeed;

	if (turn > maxTurnRad) turn = maxTurnRad;
	if (turn < -maxTurnRad) turn = -maxTurnRad;


	float nextAngle = currentAngle + turn;

	float speed = sqrtf(vx * vx + vy * vy);
	vx = cosf(nextAngle) * speed;
	vy = sinf(nextAngle) * speed;
}


void Bullet::Draw(int scaleGraph) {
	if (!flag) return;

	if (owner == OwnerType::PLAYER) {																//自機弾の色
		DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::RED), true);
		DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::WHITE), false);
	}
	else {
		BulletShape shape = GetBulletShape(type);
		BulletColor color = GetBulletColor(type);
		
		if (scaleGraph != -1 && shape != BulletShape::CIRCLE) {	//画像ハンドルが有効であり、且つ通常円(CIRCLE)以外の場合はテクスチャ回転描画
			float angle = atan2f(vy, vx);                               //進行方向の角度を算出
			float drawAngle = angle - (DX_PI_F / 2.0f);                 //画像は下向き基準なので、始線(0)からの差(-π/2)する

			int shapeIdx = static_cast<int>(shape);
			float baseR = BULLET_BASE_RADIUS[shapeIdx];
			float scale = (baseR > 0.0f) ? (r / baseR) : 1.0f;          //基準半径で割って描画倍率を判定と連動

			DrawRotaGraph((int)x, (int)y, scale, drawAngle, scaleGraph, TRUE);		//回転描画
		}
		else {
			if (color == BulletColor::RED) {
				DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::RED), true);
				DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::WHITE), false);
			}
			else {
				DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::WHITE), true);
				DrawCircle((int)x, (int)y, (int)r, ObjectColor::Get(ObjectColor::Type::BLUE), false);		//通常円(CIRCLE)や画像がない場合のフォールバック描画
			}
		}
	}
}

#pragma once
#include "Common.h"									//PlayAreaを使うためインクルード

class Bullet {
private:
	static constexpr float Margin = 3.0f;			//弾が画面外で消える判定の遊びの大きさ
	static constexpr float maxTurnRad = 1.0f;		//1フレームに曲がれる最大角度定数
	
	unsigned int color;		

public:

	enum class OwnerType{ PLAYER, ENEMY };										//誰の弾かのタイプ分け
	OwnerType owner;															//インスタンス宣言
	bool isHoming = false;														//ホーミング弾フラグ

	float charge = 0.0f;				//電荷
	float mass = 1.0f;					//質量

	float turnSpeed = 0.0f;							//弾自身の旋回しやすさを格納する変数

	float x = 0;
	float y = 0;
	float r = 0;
	float vx = 0;
	float vy = 0;
	bool flag = false;
	int type = 0;									//弾の色や見た目のタイプを示す番号
	
	
	
	bool GetFlag() {
		return flag;
	}

	void Shoot(float in_x, float in_y, float in_r, float in_vx, float in_vy, bool in_homing = false, OwnerType in_owner = OwnerType::PLAYER, float in_turnSpeed = 0.0f, int in_type = 0) {  //発射する瞬間に弾の状態を表す引数を取得させる。0.0f,0などのデフォルト定数をもつ引数は後方にまとめる。
		x = in_x;
		y = in_y;
		r = in_r;
		vx = in_vx;
		vy = in_vy;
		isHoming = in_homing;
		owner = in_owner;
		turnSpeed = in_turnSpeed;
		type = in_type;					//0:青円,1:赤円,2:青鱗弾

		flag = true;
	}

	void Guide(float tx, float ty, float turnSpeed);
		
	void Update() {
		if (!flag) return;			//flag!=trueでスキップ		
		x += vx;
		y += vy;

		using namespace PlayArea;
		if (x < Left - Margin || x > Right + Margin || y < Top - Margin || y > Bottom + Margin) flag = false;		//プレイ領域外ならばフラグを消す
		
	}

	void Draw(int scaleGraph = 0);

	void SetFlag(bool f) {
		flag = f;
	}
};

BulletManager::BulletManager() {
	for (int i = 0; i < BULLET_GRAPHIC_MAX; i++) {
		bulletGraphs[i] = -1;								//配列の中身を-1で初期化
	}

void BulletManager::ReloadGraphic() {
	
	for (int i = 0; i < BULLET_GRAPHIC_MAX; i++) {		//既存ハンドルの破棄
		if (bulletGraphs[i] != -1) {
			DeleteGraph(bulletGraphs[i]);
			bulletGraphs[i] = -1;
		}
	}

	const char* shapeFolders[] = {

void BulletManager::Update(float playerX, float playerY, float bossX, float bossY, bool bossAlive, Enemy* fairies) {
	for (int i = 0; i < PlayerBMax; i++) {
		if (!p_bullets[i].GetFlag()) continue;

		if (p_bullets[i].isHoming) {
			if (p_bullets[i].owner == Bullet::OwnerType::PLAYER) {				//OwnerTypeはBulletクラス内で定義しているので、BulletクラスのOwnerTypeであることを明示する。

				float targetX = -1.0f;
				float targetY = -1.0f;
				float minDistSq = FLT_MAX;										//FLT_MAXはfloat型が出せる最大の数字を出す
				bool isTargetFound = false;


				if (bossAlive) {																	//ボスのフラグが立ってる場合
					minDistSq = Distance::BetweenSq(bossX, bossY, p_bullets[i].x, p_bullets[i].y);	//ボスと自機弾間の距離を代入
					targetX = bossX;
					targetY = bossY;
					isTargetFound = true;															//ボスのフラグが立っているので当然見つけた判定にする
				}

				if (fairies != nullptr) {															//ヌルポインタ(妖精のデータが空っぽの状態で住所を渡してくる)時に無理矢理覗くとアクセス違反でエラーが起きるので、そうでない場合
					for (int j = 0; j < Enemy::EnemyMax; j++) {
						if (fairies[j].GetFlag()) {
							float distSq = Distance::BetweenSq(fairies[j].GetX(), fairies[j].GetY(), p_bullets[i].x, p_bullets[i].y);		//distSqに自機と妖精の距離を代入

							if (distSq < minDistSq) {												//自機に妖精の方がボスより近い場合
								minDistSq = distSq;													//一番近い距離変数に妖精の距離を代入
								targetX = fairies[j].GetX();
								targetY = fairies[j].GetY();										//ホーミングのターゲット位置も妖精のものに更新
								isTargetFound = true;
							}
						}
					}
				}

				if (isTargetFound) {
					p_bullets[i].Guide(targetX, targetY, p_bullets[i].turnSpeed);					//isTargetFoundが上のでtrueになったらホーミングのGuideに引数を渡す
				}

			}
			else if (p_bullets[i].owner == Bullet::OwnerType::ENEMY) {
				p_bullets[i].Guide(playerX, playerY, 0.05f);					//ENEMYの時は自機を狙う弾のはずなので自機座標が引数
			}
		}
		p_bullets[i].Update();
	}
	for (int i = 0; i < EnemyBMax; i++) {
		if (!e_bullets[i].GetFlag()) continue;									//フラグが立ってない場合飛ばす
		if (e_bullets[i].isHoming) {											//敵弾がホーミングする場合
			e_bullets[i].Guide(playerX, playerY, e_bullets[i].turnSpeed);
		}
		e_bullets[i].Update();
	}
}



bool BulletManager::CheckEnemyCollision(float ex, float ey, float er, Player& player) {
	for (int i = 0; i < PlayerBMax; i++) {
		if (!p_bullets[i].GetFlag()) continue;

		if (Collision::CheckCircle(ex, ey, er, p_bullets[i].x, p_bullets[i].y, p_bullets[i].r)) {		//CollisionクラスのCheckCircle関数に自機と敵弾それぞれのx,y,r値を教える
			p_bullets[i].flag = false;																	//当たった弾は消す
			
			player.AddScore(10);

			return true;
		}
	}
	return false;
}

bool BulletManager::CheckPlayerCollision(float px, float py, float pr) {
	for (int i = 0; i < EnemyBMax; i++) {
		if (!e_bullets[i].GetFlag()) continue;

		BulletShape shape = GetBulletShape(e_bullets[i].type);			//弾種を取得してshapeに格納
		float hitRate = BULLET_HIT_RATE[static_cast<int>(shape)];		//hitRateにshapeの弾種の見た目半径に対する当たり判定の比率を格納
		float enemyHitR = e_bullets[i].r * hitRate;

		if (Collision::CheckCircle(px, py, pr, e_bullets[i].x, e_bullets[i].y, enemyHitR)) {		
			e_bullets[i].flag = false;
			return true;
		}
	}
	return false;
}


void BulletManager::ConvertEnemyBulletsToItems() {
	for (int i = 0; i < EnemyBMax; i++) {
		if (e_bullets[i].GetFlag()) {

			float bx = e_bullets[i].x;
			float by = e_bullets[i].y;			//消す前に敵弾座標退避

			e_bullets[i].flag = false;			

			for (int j = 0; j < BulletManager::MAX_ITEMS; j++) {		
				if (!items[j].isExist) {
					items[j].Spawn(bx, by, ItemType::MINI_SCORE);	//小スコアアイテムを出現させる
					break;											//1つの弾をアイテムに変えたら次の弾ループへ
				}
			}
		}
	}
}

void BulletManager::LaunchItem(float x, float y, ItemType type, int count, float scatterRadius) {
	for (int i = 0; i < count; i++) {
		for (int j = 0; j < MAX_ITEMS; j++) {
			if (!items[j].isExist) {														//配列内の空いてるアイテムを探す
				float offsetX = (float)(GetRand((int)scatterRadius * 2) - scatterRadius);	
				float offsetY = (float)(GetRand((int)scatterRadius * 2) - scatterRadius);	//scatterRadiusの範囲内でランダムなオフセットを作る、半分引くことで-Rから+Rの範囲の乱数を返す

				items[j].Spawn(x + offsetX, y + offsetY, type);								//オフセットを加算した座標でアイテム呼び出し
				break;
			}
		}
	}
}

void BulletManager::UpdateItems(Player& player, float collectRadius, bool isAutoCollect) {
	for (int i = 0; i < BulletManager::MAX_ITEMS; i++) {
		if (items[i].isExist) {
			if (isAutoCollect && !items[i].isTargeted) {	//上部回収フラグオンで、まだ回収モードに成ってないアイテムがある場合
				items[i].isTargeted = true;						//アイテムの回収モードON
			}
			items[i].Update(player, collectRadius);
		}
	}
}

void BulletManager::DrawItems() {
	for (int i = 0; i < BulletManager::MAX_ITEMS; i++) {
		if (items[i].isExist) {
			items[i].Draw();
		}
	}
}

void BulletManager::Draw() {
	for (int i = 0; i < PlayerBMax; i++) {
		if (p_bullets[i].GetFlag() != 0) {
			p_bullets[i].Draw();
		}
	}
	for (int i = 0; i < EnemyBMax; i++) {
		if (!e_bullets[i].GetFlag()) continue;

		e_bullets[i].Draw(bulletGraphs[e_bullets[i].type]);
	}
}

	void AimedShot(float ex, float ey, float sr, float speed, const Player& player, BulletManager* bm, int graphicType) {
		float dx = player.GetX() - ex;
		float dy = player.GetY() - ey;									//敵から自機への相対位置ベクトル(dx,dy)を算出

		float angleRad = atan2f(dy, dx);								//敵から自機の角度[rad]を算出
		float angleDeg = angleRad * 180.0f / DX_PI_F;					//弧度法→度数法変換
		NWayShot(ex, ey, sr, speed, 1, 0.0f, angleDeg, bm, graphicType);				//NWayShot関数へ処理を任せる
	}

	void NWayShot(float ex, float ey, float sr, float speed, int way, float totalAngleDeg, float baseAngleDeg, BulletManager* bm, int graphicType) {
		if (way <= 1) {													//way数が1以下の場合の零除算防止
			float rad = baseAngleDeg * DX_PI_F / 180.0f;				//度数法→弧度法変換
			float vx = cosf(rad) * speed;								//弾速のx軸成分
			float vy = sinf(rad) * speed;								//弾速のy軸成分
			bm->LaunchEnemyBullet(ex, ey, sr, vx, vy, false, 0.0f, graphicType);		//インスタンスbmに発射依頼
			return;
		}

		const float DEG2RAD = DX_PI_F / 180.0f;							//度数法→弧度法変換の係数

		float stepAngleDeg = totalAngleDeg / (way - 1);					//弾と弾の間の角度(step角)の計算
		float startAngleDeg = baseAngleDeg - (totalAngleDeg / 2.0f);	//発射を開始する最初(一発目)の角度

		if (totalAngleDeg >= 360.0f) {
			stepAngleDeg = totalAngleDeg / way;							//全方位の時は最初と最後の弾が重なるので調整する
		}

		for (int i = 0; i < way; i++) {
			float currentAngleDeg = startAngleDeg + (stepAngleDeg * i);	//ループごとの弾の角度を計算する
			float rad = currentAngleDeg * DEG2RAD;						//弧度法へ変換

			float vx = cosf(rad) * speed;
			float vy = sinf(rad) * speed;

			bm->LaunchEnemyBullet(ex, ey, sr, vx, vy, false, 0.0f, graphicType);
		}
	}


	void ExecShot(float ex, float ey, const ShotConfig& conf, const Player& player, BulletManager* bm) {
		switch (conf.type) {
		case PT::Aimed:
			AimedShot(ex, ey, conf.radius, conf.speed, player, bm, conf.graphicType);
			break;

		case PT::NWay:
		{
			float dx = player.GetX() - ex;
			float dy = player.GetY() - ey;
			float aimAngleDeg = atan2f(dy, dx) * 180.0f / DX_PI_F;

			NWayShot(ex, ey, conf.radius, conf.speed, conf.way, conf.totalAngle, aimAngleDeg, bm, conf.graphicType);
			break;
		}

		case PT::RotateAll:
			NWayShot(ex, ey, conf.radius, conf.speed, conf.way, 360.0f, conf.baseAngle, bm, conf.graphicType);
			break;
		}
	}
}

void BulletTestScene::UpdateParams() {
	if (Input::IsTrigger(KEY_INPUT_TAB)) {
		isTuningMode = !isTuningMode;
	}

	if (!isTuningMode || params.empty()) return;

	// カーソル上下でパラメータ選択
	if (Input::IsTrigger(KEY_INPUT_UP)) {
		currentParamIdx = (currentParamIdx - 1 + (int)params.size()) % (int)params.size();
	}
	if (Input::IsTrigger(KEY_INPUT_DOWN)) {
		currentParamIdx = (currentParamIdx + 1) % (int)params.size();						//↑↓でパラメータの種類選択
	}

	// 左右キー（または長押し）で値変更
	auto& p = params[currentParamIdx];
	if (Input::IsPress(KEY_INPUT_LEFT)) {
		p.value -= p.step;
		if (p.value < p.minVal) p.value = p.minVal;
	}
	if (Input::IsPress(KEY_INPUT_RIGHT)) {
		p.value += p.step;
		if (p.value > p.maxVal) p.value = p.maxVal;											//←→で値変更
	}
}

		bm->ClearEnemyBullets();
	}
	
	float currentI		= params[0].value;
	float fluxK			= params[1].value;
	float softeningR	= params[2].value;
	float maxBzCap		= params[3].value;
	float shotSpeed		= params[4].value;
										
	if (testState == 0) {										//弾幕切替の状態が0の場合
		if (testTimer % 60 == 0) {
			float aimAngle = atan2f(testPlayer.GetY() - originY, testPlayer.GetX() - originX) * 180.0f / DX_PI_F;
			BulletPattern::NWayShot(originX, originY, 4.0f, shotSpeed, 9, 180.0f, aimAngle, bm, 0); //青(負電荷)
			BulletPattern::NWayShot(originX, originY, 4.0f, shotSpeed, 9, 180.0f, aimAngle, bm, 1); //赤(正電荷)
		}
	}
	else if (testState == 1) {
		if (testTimer % 5 == 0) {
			float baseAngleDeg = (testTimer * 1.5f);												//回転速度
			int colorType = (testTimer % 10 == 0) ? 1 : 0;
			BulletPattern::NWayShot(originX, originY, 4.0f, shotSpeed, 8, 360.0f, baseAngleDeg, bm, colorType);
		}
	}

	bm->Update(testPlayer.GetX(), testPlayer.GetY(), originX, originY, true, nullptr);		//敵の配列はダミーなのでヌルポインタを渡す

	for (int i = 0; i < bm->GetEnemyBulletMax(); i++) {
		Bullet& b = bm->GetEnemyBullet(i);
		if (!b.GetFlag()) continue;

		if (b.type == 0) b.charge = -1.0f;
		else if (b.type == 1) b.charge = 1.0f;
		else b.charge = 0.0f;

		if (b.charge != 0.0f && b.mass > 0.0f) {
			float currentSpeed = sqrtf(b.vx * b.vx + b.vy * b.vy);
			if (currentSpeed <= 0.001f) continue;

			//線電流方向ベクトル u
			float u_x = laserDirX;
			float u_y = laserDirY;

			//弾の位置ベクトル (レーザー始点基準)
			float px = b.x - laserStartX;
			float py = b.y - laserStartY;

			//直線への射影長 t
			float t = px * u_x + py * u_y;

			//直線上の最近接点
			float closestX = laserStartX + t * u_x;
			float closestY = laserStartY + t * u_y;

			//最近接点から弾への法線ベクトル n
			float nx = b.x - closestX;
			float ny = b.y - closestY;

			//距離 r の算出とソフトニング
			float distSq = nx * nx + ny * ny;
			float softeningSq = softeningR * softeningR;
			float rEff = sqrtf(distSq + softeningSq);

			//単位法線ベクトル
			float rReal = sqrtf(distSq);
			if (rReal > 0.001f) {
				nx /= rReal;
				ny /= rReal;
			}
			else {
				nx = -u_y;
				ny = u_x;
			}

			//磁束密度 B_z = (K * I / r) * (u × n)_z
			float Bz = (fluxK * currentI / rEff) * (u_x * ny - u_y * nx);

			//磁束密度の上限クランプ
			if (Bz > maxBzCap) Bz = maxBzCap;
			if (Bz < -maxBzCap) Bz = -maxBzCap;

			//ローレンツ力 F = q(v × B)
			float Fx = b.charge * (b.vy * Bz);
			float Fy = b.charge * (-b.vx * Bz);

			//速度ベクトルの偏向
			b.vx += Fx / b.mass;
			b.vy += Fy / b.mass;

			//磁束密度による仕事はゼロのため、元の速さに再正規化（純粋旋回）
			float newSpeed = sqrtf(b.vx * b.vx + b.vy * b.vy);
			if (newSpeed > 0.001f) {
				b.vx = (b.vx / newSpeed) * currentSpeed;
				b.vy = (b.vy / newSpeed) * currentSpeed;
			}
		}
	}

	if (Input::IsTrigger(KEY_INPUT_X) && testPlayer.GetBomb() > 0) {				//ボム・喰らいボムの処理
		if (testPlayer.IsDeathTimerActive()) {			//喰らいボムタイマがアクティブの場合
			testPlayer.StopDeathTimer();									//喰らいボムタイマストップ
			testPlayer.SetInVincible(testPlayer.GetInvincibleTime());		//点滅
			bm->ConvertEnemyBulletsToItems();								//敵弾を小スコアアイテム変換
			testPlayer.ComsumeBomb();										//ボム消費
		}
		else if (!testPlayer.IsInvincible()) {			//点滅していない時
			bm->ConvertEnemyBulletsToItems();		//敵弾を小スコアアイテム変換
			testPlayer.ComsumeBomb();				//ボム消費
		}
	}


	if (!testPlayer.IsInvincible()) {		//点滅(無敵)中でない場合
		bool isHit = bm->CheckPlayerCollision(testPlayer.GetX(), testPlayer.GetY(), testPlayer.GetHitR());	//当たり判定
		if (isHit && !testPlayer.IsDeathTimerActive()) {			
			testPlayer.StartDeathTimer();						//被弾して且つ、喰らいボムまでのタイマがインアクティブな場合に喰らいまでのボムタイマ起動
		}
	}

	if (testPlayer.GetLife() < 0) {
	/*	testPlayer.Reset(TRUE);									//残機数が負に成ったらデフォ(TRUE:プラクティス)の残機数をセットし直して何度でも被弾可能とする
		bm->ClearAllBullets();									//復帰時の被弾ループ回避のために全敵弾消去	*/	
		return true;
	}
	return false;
}

#pragma once

class BurstController {
private:
	int remain = 0;								//スロット内の残り回数
	int timer = 0;								//スロット内の次までの待ち時間
	int dataIdx = 0;							//参照するデータの番号

public:
	void Activate(int count, int idx = 0) {
		remain = count;
		timer = 0;
		dataIdx = idx;
	}

	bool Tick(int interval) {
		if (remain <= 0)return false;

		if (timer <= 0) {						//次までの待ち時間が0以下の場合
			timer = interval;					//次までの待ち時間をリセット
			remain--;							//残り回数をデクリメント
			return true;						
		}

		timer--;								//待ち時間をデクリメント
		return false;
	}

	bool IsActive() const { return remain > 0; }
	int GetDataIdx() const { return dataIdx; }
	void Reset() { remain = 0; timer = 0; }
};

namespace MyMath {
	inline void AccelarateToTarget(float targetX, float targetY, float currentX, float currentY, float& vx, float& vy, float acceleration, float speedCap) {		//2点間の誘導加速及び速度制限を行う関数
		float dx = targetX - currentX;
		float dy = targetY - currentY;
		float dist = sqrtf(dx * dx + dy * dy);

		if (dist > 0.0f) {
			vx += (dx / dist) * acceleration;
			vy += (dy / dist) * acceleration;
		}

		float currentSpeed = sqrtf(vx * vx + vy * vy);
		if (currentSpeed > speedCap) {
			vx = (vx / currentSpeed) * speedCap;
			vy = (vy / currentSpeed) * speedCap;
		}
	}
}

/**
	 * @brief 全方位回転弾 (PT::RotateAll) の交差用回転速度 (rotSpeed) をコンパイル時に自動計算する
	 *
	 * --- RotateAllの回転速度(rotSpeed)算出メモ ---
	 * 1回の発射で進む角度をΔθ[deg]としたとき、
	 * 全方位弾の隙間(step = 360 / way)に対してΔθがstepの整数倍になると、
	 * 前の弾の軌道と重なって回転して見えなくなる。
	 *
	 * 綺麗にバラけさせるには、
	 *   step = 360.0f / way
	 *   Δθ   = step * (phaseShift + k)
	 * となるように rotSpeed を定めれば良い。
	 *
	 *   rotSpeed = (Δθ * π) / (si * 180)
	 *
	 * @param way 弾の全方位way数
	 * @param si 発射間隔 (interval[F])
	 * @param phaseShift 位相ズレ比率 (1.0f/2.0f: 半周期ズレ, 1.0f/3.0f: 1/3周期ズレ, 1.0f/4.0f: 1/4周期ズレ)
	 * @param k 回転感を出すための追加周回数 (0,1,2...)
	 * @return constexpr float 算出された rotSpeed[rad/F]
	 */
constexpr float CalcRotateSpeed(int way, int si, float phaseShift = (1.0f / 3.0f), int k = 2) {
	float step = 360.0f / static_cast<float>(way);
	float deltaTheta = step * (phaseShift + static_cast<float>(k));
	return (deltaTheta * 3.14159265f) / (static_cast<float>(si) * 180.0f);
}

		if (in_dropCounts != nullptr) {
			m_dropCounts[i] = in_dropCounts[i];
		}
		else {
			m_dropCounts[i] = 0;
		}
	}
}

void Enemy::Update(const Player& player, BulletManager* bm) {	//playerからはパラメータを参照するだけなので書き換えないconstと参照の&で、bmは書き換えたりするので*を使う
	using namespace PlayArea;
	x += vx;
	y += vy;
	if (Left - r > x || Right + r<x || Top - r>y || Bottom + r < y) flag = false;		//画面外に行ったら消える

	if (ShotTimer != 0 && ShotTimer % m_shotConf.interval == 0) {
		BulletPattern::ExecShot(x, y, m_shotConf, player, bm);
	}
	ShotTimer++;
}

void Enemy::Draw() {
	if (flag) {
		DrawCircle(x, y, r, GetColor(0, 0, 255), true);
	}
}

bool Enemy::CheckCollision(Player& player, BulletManager* bm) {
	if (!flag) return false;											//フラグの立ってない敵は当たらない
	
	for (int i = 0; i < bm->GetPlayerBulletMax(); i++) {
		Bullet& b = bm->GetPlayerBullet(i);								//自機弾各々が持つ検体番号を格納
		if (!b.GetFlag()) continue;										//自機フラグが立っていない時飛ばす

		if (Collision::CheckCircle(x, y, r, b.x, b.y, b.r)) {			//
			hp -= 1;								//hpをデクリメント
			b.SetFlag(false);						//弾を消す

			if (bm->CheckEnemyCollision(x, y, r, player)) {}

			if (hp <= 0) {
				flag = false;							//妖精消す
				for (int typeIdx = 0; typeIdx < DROP_TYPE_MAX; typeIdx++) {
					int count = m_dropCounts[typeIdx];
					if (count > 0) {
						ItemType iType = static_cast<ItemType>(typeIdx);
						bm->LaunchItem(x, y, iType, count, ItemScatterRadius);
					}
				}
				return true;							//被弾判定を返す
			}
		}
	}
	return false;
}

int FPSManager::startTime = 0;
int FPSManager::goalTime = 0;
int FPSManager::count = 0;
float FPSManager::fps = 0.0f;

void FPSManager::Update() {
	if (count == 0) {
		startTime = GetNowCount();						//countが0の時パソコン起動時からの経過時間[ms]を格納
	}

	count++;
	
	if (count == N) {
		goalTime = GetNowCount();						//countが60になった時に計測終了時間を格納
		fps = (1000.0f * N) / (goalTime - startTime);	//1秒辺り何フレーム進むかを比例式から算出
		count = 0;										//60になってfps算出終わったら初期化して再度算出をする
	}
}

void FPSManager::Draw() {
	DrawFormatString(520, 460,GetColor(255, 255, 255),"FPS: %.1f",fps);		//Cのprintfと同じ機能の関数。DrawStringでは全て固定の文字を直接出力なので不適。
}

void FPSManager::Wait() {
	int idealTime = count * 1000 / N;										//本来経過しているべき理想msを算出
	int actualTime = GetNowCount() - startTime;								//実際に掛かったmsを算出

	int waitTime = idealTime - actualTime;									
	if (waitTime > 0) {
		WaitTimer(waitTime);												//idealTimeよりactualTimeが早かったら、waitTimeだけ待機
	}
}

	SetDrawScreen(gh);

			unsigned int color = GetCustomPixel(category, typeIdx, y, x);
			int r, g, b;
			GetColor2(color, &r, &g, &b);

void Input::Update() {
	char tmpKey[256];
	GetHitKeyStateAll(tmpKey);					//256キー状態を一括取得

	for (int i = 0; i < 256; i++) {				
		
		oldKeys[i] = keys[i];		//tmpKeyで上書きする前に現在の状態をoldとして保存
		if (tmpKey[i] != 0) {		
			keys[i]++;							//押されているキーを見つけて押されているならフレームを加算
		}
		else {
			keys[i] = 0;						//離されたら0にリセット
		}
	}

bool Input::IsTrigger(int keyCode) {
	return keys[keyCode] == 1;					//押されて1フレーム目のみtrueを返す
}

bool Input::IsRelease(int keyCode) {
	return oldKeys[keyCode] > 0 && keys[keyCode] == 0;		//前フレームで押下(>0)かつ今フレームで離されている(==0)
}

	this->y = startY;
	this->type = itemType;
	this->isExist = true;
	this->isTargeted = false;
	this->topDelayTimer = 0;

	this->gravity = DEFAULT_GRAVITY;

	if (itemType == ItemType::MINI_SCORE) {
		this->vy = MINI_SCORE_SPAWN_VY;
	}
	else if (itemType == ItemType::ZANKI) {
		this->vy = ZANKI_SPAWN_VY;
	}
	else {
		this->vy = BASE_SPAWN_VY;
	}

void Item::Update(Player& player, float collectRadius) {
	if (!isExist) return;			//存在フラグが立っていないならスキップ

	float playerX = player.GetX();
	float playerY = player.GetY();

	vy += gravity;					//加速

	if (isTargeted) {			//回収モード
		float toPlayerDist = sqrtf((playerX - x) * (playerX - x) + (playerY - y) * (playerY - y));	//自機とアイテムの距離

		if (toPlayerDist < ADSORPTION_DISTANCE_LIMIT) {
			x = playerX;
			y = playerY;
			isExist = false;

			switch (type) {
			case ItemType::MINI_SCORE:	player.AddScore(SCORE_VALUE_MINI);	break;		//小スコア： +10点
			case ItemType::SCORE:		player.AddScore(SCORE_VALUE_NORMAL);	break;		//　スコア： +1000点
			case ItemType::POWER:		player.AddPower(POWER_GAIN_NORMAL);		 player.AddScore(SCORE_VALUE_POWER);		break;		
			case ItemType::ZANKI:		player.Extend();		break;
			case ItemType::BOMB:		player.AddBomb();		break;
			case ItemType::BIG_POWER:	player.AddPower(POWER_GAIN_BIG);		 player.AddScore(SCORE_VALUE_POWER);		break;
			}
		}
		else {						//距離が離れてるならすり抜けない速度(BASE_SPEED_CAP)で自機を追尾
			x += ((playerX - x) / toPlayerDist) * BASE_SPEED_CAP;
			y += ((playerY - y) / toPlayerDist) * BASE_SPEED_CAP;
		}

		if (Collision::CheckCircle(playerX, playerY, PLAYER_MIN_HIT_R, x, y, r)) {
			isExist = false;

			switch (type) {
			case ItemType::MINI_SCORE:	player.AddScore(10);	break;		//小スコア： +10点
			case ItemType::SCORE:		player.AddScore(1000);	break;		//　スコア： +1000点
			case ItemType::POWER:		player.AddPower(1);		break;		//　パワー： +1
			case ItemType::ZANKI:	break;
			case ItemType::BOMB:	break;
			}
		}
	}
	else {							//回収モードではない時
		if (vy < 0.0f) {
			vy += gravity;			//加速
			y += vy;				//位置を加算
		}
		else {
			if (type == ItemType::MINI_SCORE) {
				topDelayTimer++;

				if (topDelayTimer >= MINI_SCORE_DELAY_FRAME) {
					isTargeted = true;
				}
			}
			else {
				vy += gravity;
				if (vy > fallSpeed) {
					vy = fallSpeed;
				}
				y += vy;

				if (Collision::CheckCircle(playerX, playerY, collectRadius, x, y, r)) {
					isTargeted = true;
				}
			}
		}
		if (y > PlayArea::Bottom + PLAYAREA_BOTTOM_MARGIN) {	//下端の遊びを過ぎたらフラグ折り
			isExist = false;
		}
	}
}

	int targetTypeIdx = static_cast<int>(type);
	double scale = 1.0;

		scale = BIG_POWER_SCALE;											//大パワーはパワーの画像を1.5倍に拡大して描画
	}

	int gh = GetCustomGraphHandle(EditCategory::ITEM, targetTypeIdx);
	if (gh != -1) {
		DrawRotaGraph((int)x, (int)y, scale, 0.0, gh, TRUE);
	}
	else {
		unsigned int color;
		int size = 5;

		int typeIdx = 0;
		switch (type) {
		case ItemType::MINI_SCORE:
			color = ObjectColor::Get(ObjectColor::Type::CYAN);    size = 3; break;		//小スコア

int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow) {
	bool isWindow = true;								//ウィンドウかフルスクかの状態を保存するフラグ
	ChangeWindowMode(TRUE);								//初期状態はウィンドウモード

	SetWaitVSyncFlag(FALSE);							//DxLib標準機能のブレーキ機能フラグを消して、自作時間制御システムで管理
	SetDrawScreen(DX_SCREEN_BACK);						//裏描画
	if (DxLib_Init() == -1) return -1;					//エラーが発生したら終了

	HWND hwnd = GetMainWindowHandle();								//GetMainWindowHandle()はDxLibが内部で作成したHWNDを取得する関数
	OldWndProc = (WNDPROC)SetWindowLongPtr(hwnd, GWLP_WNDPROC, (LONG_PTR)MyWndProc);		//SetWindowLongPtr の戻り値（元の WndProc）を OldWndProc に保存する

	BulletManager bm;
	ScoreManager scoreManager;
	bool isScoreLoaded = scoreManager.Load();
	StageManager stageManager;
	HUD hud;
	hud.Init();
	InitCustomGraphics();		

	SceneManager sceneManager(&bm, &scoreManager, &stageManager, &hud, isScoreLoaded);		//実体のポインタをSceneManagerに渡して初期化


	while (ProcessMessage() == 0 && ClearDrawScreen() == 0) {								//メインループ
		Input::Update();		
		FPSManager::Update();																//fps計算の更新

		if (sceneManager.CanToggleFullscreen() && Input::IsTrigger(KEY_INPUT_F)) {			//名前入力シーンではFが入力出来るように除外
			isWindow = !isWindow;													//trueとfalseの状態を交互に交換する
			ChangeWindowMode(isWindow ? TRUE : FALSE);

			bm.ReloadGraphic();
			stageManager.ReloadGraphic();
			hud.ReloadFont();
			RefreshAllCustomGraphs();
			SetDrawScreen(DX_SCREEN_BACK);
		}
		
		sceneManager.Update();
		sceneManager.Draw();

		FPSManager::Draw();																		//fpsの描画。ゲームのどの描画よりも前に描画する為ScreenFlipの直前に記述
		DxLib::ScreenFlip();
		FPSManager::Wait();																		//fps固定
	}
	DxLib::DxLib_End();
	return 0;
}

void Player::Update(BulletManager* bm) {
	using namespace PlayArea;
	int vx = 0;
	int vy = 0;
	move_v = para.vn;												//毎フレームで通常速度に初期化することで低速状態を解除

	if (Input::IsPress(KEY_INPUT_DOWN)) vy = 1;
	if (Input::IsPress(KEY_INPUT_UP)) vy = -1;
	if (Input::IsPress(KEY_INPUT_RIGHT)) vx = 1;
	else if (Input::IsPress(KEY_INPUT_LEFT)) vx = -1;				//ここでelse ifにすることでRIGHTの方の判定を強くして妖々夢らしくする
	if (Input::IsPress(KEY_INPUT_LSHIFT)|| Input::IsPress(KEY_INPUT_RSHIFT)) {
		move_v = para.vs;										//低速
	}

	float speed = move_v;
	if (vx != 0 && vy != 0) speed /= sqrtf(2.0f);				//斜め移動の正規化

	x += vx * speed;
	y += vy * speed;

	if (x < Left + para.areaW) x = Left + para.areaW;			//x - para.areaW < Leftの左辺をxのみにするために移項した
	if (x > Right - para.areaW) x = Right - para.areaW;
	if (y < Top + para.areaH) y = Top + para.areaH;				//xと同様
	if (y > Bottom - para.areaH) y = Bottom - para.areaH;


	if (invincibleTimer > 0) {
		invincibleTimer--;										//無敵時間の変数が0超過ならデクリメント
	}

	bool isPress = (Input::IsPress(KEY_INPUT_Z) || Input::IsPress(KEY_INPUT_SPACE));		//ショット発射ボタンを押しているか
	bool isSlow = (move_v == para.vs);																	//低速状態か

	if (isPress && needle_shot_timer <= 0) {							//発射ボタン押下されている状態且つショットクールタイムが0の場合
		if (mCurrentWeapon != nullptr) {
			ReimuB_Weapon* rw = dynamic_cast<ReimuB_Weapon*>(mCurrentWeapon);
			if (rw != nullptr) {
				rw->ActivateNeedle();
				needle_shot_timer = rw->GetNeedleSca(mDisplayPower);
			}
			//mCurrentWeapon->Fire(*this, bm);
		}
		if (!isSlow) {											//低速状態ではない場合
			ActivateWeapon(SubSlot, SubHoming.L);				//サブショット(ホーミング弾追加)
		}
		//shot_timer = sca;										//長押し時のクールタイム
	}
	else if (!isPress && needle_shot_timer > 0) {				
		//shot_timer = scm;										//連打時のクールタイムに上書きする
		if(mCurrentWeapon != nullptr) {
			ReimuB_Weapon* rw = dynamic_cast<ReimuB_Weapon*>(mCurrentWeapon);
			if (rw != nullptr) {
				int targetScm = rw->GetNeedleScm(mDisplayPower);
				if (needle_shot_timer > targetScm) {
					needle_shot_timer = targetScm;
				}
			}
		}
	}

	if (isPress && zabuton_shot_timer <= 0) {
		if (mCurrentWeapon != nullptr) {
			ReimuB_Weapon* rw = dynamic_cast<ReimuB_Weapon*>(mCurrentWeapon);
			if (rw != nullptr) {
				rw->ActivateZabuton();
				zabuton_shot_timer = rw->GetZabutonSca(mDisplayPower);
			}
		}
	}
	else if (!isPress && zabuton_shot_timer > 0) {
		if (mCurrentWeapon != nullptr) {
			ReimuB_Weapon* rw = dynamic_cast<ReimuB_Weapon*>(mCurrentWeapon);
			if (rw != nullptr) {
				int targetScm = rw->GetZabutonScm(mDisplayPower);
				if (zabuton_shot_timer > targetScm) {
					zabuton_shot_timer = targetScm;
				}
			}
		}
	}


	if (mCurrentWeapon != nullptr) {
		mCurrentWeapon->Update(*this, bm);						//ウェポンクラス側のスロット更新
	}

	/*

	for (int s = 0; s < SLOT_MAX; s++) {														//画面上に同時に存在できるスロット数繰り返す
		if (MainSlot[s].Tick(MainShot.LI)) {
			bm->LaunchPlayerBullet(x - MainShot.sox, y, MainShot.sr, MainShot.ssx, MainShot.ssy, false, 0.0f);		//ここでturnSpeedにPlayerHomingPowerを代入する。
			bm->LaunchPlayerBullet(x + MainShot.sox, y, MainShot.sr, MainShot.ssx, MainShot.ssy, false, 0.0f);
		}
	}

	*/

	for (int s = 0; s < SLOT_MAX; s++) {
		if (SubSlot[s].Tick(SubHoming.LI)) {
			bm->LaunchPlayerBullet(x, y, SubHoming.sr, 0.0f, SubHoming.ssy, true, SubHoming.turn);
		}
	}

	//if (shot_timer > 0) shot_timer--;

	if (needle_shot_timer > 0) needle_shot_timer--;
	if (zabuton_shot_timer > 0) zabuton_shot_timer--;

	if (IsDeathTimerActive()) {				//喰らいボム受付時間中の場合
		DecrementDeathTimer();							//deathTimerのデクリメント
		if (!IsDeathTimerActive()) {					//タイマが0に成った瞬間
			if (mLife > 0) {					//残機がある場合
				LoseLife();								//残機を減らす				
				loadingBomb();							//基準ボム数の装填
				SetInVincible(GetInvincibleTime());		//無敵状態遷移
				bm->ClearEnemyBullets();				//敵弾消去
			}
			else {
				mLife = -1;				//メインループでSCENE_GAMEOVERに移行する為にmLife=-1にしておく
			}
		}
	}
	if (mDisplayScore < mScore) {
		int diff = mScore - mDisplayScore;					//diff：実際のスコアと表示スコアの差
		if (diff > SCORE_ROLLUP_LARGE_THRESHOLD) {			
			mDisplayScore += SCORE_ROLLUP_LARGE_STEP;		//diff > THRESHOLD(差が大きい)の場合、一気に加算
		}
		else {
			mDisplayScore += SCORE_ROLLUP_SMALLSTEP;		//基本はSMALLSTEPで加算
		}

		if (mDisplayScore > mScore) {
			mDisplayScore = mScore;							//加算しすぎて越えたら実際のスコアに強制的に戻す
		}
	}

	if (mDisplayPower < mPower) {								
		mDisplayPower++;										//実際のパワーが表示よりも大きいときはインクリメント
		if (mDisplayPower > mPower) mDisplayPower = mPower;		//越えたら戻す
	}
}


void Player::Reset(bool isPractice) {
	x = (float)PlayArea::DefaultPlayerX;				//自機x座標初期化
	y = (float)PlayArea::DefaultPlayerY;				//自機y座標初期化

	if (mCurrentWeapon != nullptr) {
		mCurrentWeapon->Reset();				//外部インスタンス側でもウェポンスロットの初期化
	}

	for (int i = 0; i < SLOT_MAX; i++) {
		//MainSlot[i].Reset();							//スロット内の残り弾数やら次までの残り時間を初期化
		SubSlot[i].Reset();								//上と同様
	}
	needle_shot_timer = 15;
	zabuton_shot_timer = 15;										//ショットのクールタイムも初期化
	invincibleTimer = 0;

	mLife = (isPractice ? stdPracLife : stdLife);			//プラクティスモードではstdPracLife,通常モードならstdLifeを初期残機として代入
	mBomb = stdBomb;										//初期残ボム数

	if (isPractice) {			//プラクティスモード
		mScore = 0;
		mDisplayScore = 0;
		mPower = POWER_MAX;				//プラクティスでは最初からパワー最大
		mDisplayPower;
	}
	else {						//通常スタート
		mScore = 0;
		mDisplayScore = 0;
		mPower = 0;
		mDisplayPower = 0;
	}
}

	}

	int drawHandle = GetCustomGraphHandle(EditCategory::PLAYER, 0);
	DrawRotaGraph((int)x, (int)y, para.drawScale, 0.0f, drawHandle, TRUE);
}

	}
}

void ReimuB_Weapon::ActivateNeedle() {			//安全に針用の空きスロットを探して起動するインターフェース
	for (int i = 0; i < SLOT_MAX; i++) {
		if (!NeedleSlot[i].IsActive()) {
			NeedleSlot[i].Activate(BURST_COUNT);
			break;
		}
	}
}

void ReimuB_Weapon::ActivateZabuton() {			//安全に座布団用の空きスロットを探して起動するインターフェース
	for (int i = 0; i < SLOT_MAX; i++) {
		if (!ZabutonSlot[i].IsActive()) {
			ZabutonSlot[i].Activate(BURST_COUNT);
			break;
		}
	}
}



void ReimuB_Weapon::Fire(const Player& player, BulletManager* bm) {
	for (int i = 0; i < SLOT_MAX; i++) {				//針スロットの秋を探してアクティベート
		if (!NeedleSlot[i].IsActive()) {
			NeedleSlot[i].Activate(BURST_COUNT);
			break;
		}
	}
	
	for (int i = 0; i < SLOT_MAX; i++) {				//座布団スロットも同様
		if (!ZabutonSlot[i].IsActive()) {
			ZabutonSlot[i].Activate(BURST_COUNT);
			break;
		}
	}
}

void ReimuB_Weapon::Update(const Player& player, BulletManager* bm) {
	WeaponPhaseConfig cfg = GetConfig(player.GetDisplayPower());

	for (int s = 0; s < SLOT_MAX; s++) {				
		if (NeedleSlot[s].IsActive() && NeedleSlot[s].Tick(cfg.needleLI)) {		//針スロットのLIごとにスロット分撃つ
			for (int i = 0; i < cfg.needleCount; i++) {
				const auto& b = cfg.needles[i];
				bm->LaunchPlayerBullet(player.GetX() + b.offsetX, player.GetY() + b.offsetY, BULLET_RADIUS, b.speedX, b.speedY, IS_HOMING, TURN_SPEED);
			}
		}

		if (ZabutonSlot[s].IsActive() && ZabutonSlot[s].Tick(cfg.zabutonLI)) {	//座布団も同様
			for (int i = 0; i < cfg.zabutonCount; i++) {
				const auto& b = cfg.zabutons[i];
				bm->LaunchPlayerBullet(player.GetX() + b.offsetX, player.GetY() + b.offsetY, BULLET_RADIUS, b.speedX, b.speedY, IS_HOMING, TURN_SPEED);
			}
		}
	}
}

static uint32_t GetCurrentDateAsInt() {
	std::time_t t = std::time(nullptr);		//std::time：C/C++の標準時計関数、1970年1月1日からの経過秒数をtへ格納
	std::tm now;							//年月日時分秒に分解して保持するための標準構造体std::tmの変数now
	localtime_s(&now, &t);					//秒数データtをJSTの年月日に変換してnowに書き込む

	uint32_t year = static_cast<uint32_t>(now.tm_year + 1900);	//1900年からの経過年数が入っているので1900を加算
	uint32_t month = static_cast<uint32_t>(now.tm_mon + 1);		//0~11で月を表しているので1を加算
	uint32_t day = static_cast<uint32_t>(now.tm_mday);			//日にちを抽出してdayに格納

	return (year * 10000) + (month * 100) + day;				//年月日を8桁に結合する計算
}

void SceneManager::Update() {
	switch (scene) {
	case SCENE_TITLE:
		if (Input::IsTrigger(KEY_INPUT_UP)) {
			TitleMenu.Prev();
		}
		if (Input::IsTrigger(KEY_INPUT_DOWN)) {
			TitleMenu.Next();
		}
		if (Input::IsTrigger(KEY_INPUT_T)) {
			m_bm->ClearAllBullets();
			m_bm->ClearAllItems();
			playingBulletTest = true;				//テストシーンであることを示すフラグ
			testScene.Init();
			scene = SCENE_BULLET_TEST;
		}

		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			int select = TitleMenu.GetSelect();											//上で計算し終わって決定する直前のmenu_cursorを利用
			if (select == 0) {
				lastSelectedStage = 1;
				lastSelectedPhase = 0;
				m_bm->ClearAllBullets();
				m_bm->ClearAllItems();
				playingPractice = false;											//通常スタート時はフラグを折る
				playingBulletTest = false;											//同様にテストシーンフラグを折る
				m_stageManager->Init(m_bm, lastSelectedStage, lastSelectedPhase, playingPractice);
				scene = SCENE_STAGE;
			}
			if (select == 1) {
				scene = SCENE_PRACTICE;
			}
			if (select == 2) scene = SCENE_RANKING;
			if (select == 3) scene = SCENE_QUIT_CONFIRM;
			if (select == 4) scene = SCENE_QUIT_CONFIRM;
		}
		if (Input::IsTrigger(KEY_INPUT_ESCAPE)) {
			TitleMenu.SetSelect(4);
		}
		break;



	case SCENE_BULLET_TEST:
	{
		bool isGameOver = testScene.Update(m_bm);
		if (isGameOver) {
			GameOverMenu.SetSelect(0);		//ゲームオーバーメニューでのカーソルを初期化
			scene = SCENE_GAMEOVER;
		}
		if (Input::IsTrigger(KEY_INPUT_ESCAPE)) {
			m_bm->ClearAllBullets();
			playingBulletTest = false;		//抜ける時は弾全消しして、テストシーンフラグを折る
			scene = SCENE_TITLE;
		}
	}
	break;



	case SCENE_PRACTICE:
		if (Input::IsTrigger(KEY_INPUT_UP)) PracticeMenu.Prev();
		if (Input::IsTrigger(KEY_INPUT_DOWN)) PracticeMenu.Next();
		if (Input::IsPress(KEY_INPUT_ESCAPE)) scene = SCENE_TITLE;

		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			lastSelectedStage = 1;
			lastSelectedPhase = PracticeMenu.GetSelect();
			playingPractice = true;
			playingBulletTest = false;									//プラクティススタート時はテストフラグを折る

			m_bm->ClearAllBullets();
			m_bm->ClearAllItems();
			m_stageManager->Init(m_bm, lastSelectedStage, lastSelectedPhase, playingPractice);
			scene = SCENE_STAGE;
		}
		break;



	case SCENE_STAGE:
	{
		m_stageManager->Update(m_bm);

		m_bm->Update(
			m_stageManager->GetPlayer().GetX(),
			m_stageManager->GetPlayer().GetY(),
			m_stageManager->GetBoss().GetX(),
			m_stageManager->GetBoss().GetY(),
			m_stageManager->GetBoss().GetFlag(),			//stageManagerからplayerとBossを借りてきて座標を渡す
			m_stageManager->GetFairies()
		);

		bool isAutoCollect = (m_stageManager->GetPlayer().GetY() < PlayArea::ItemCollectionLine);						//自機が上部回収ラインより上の時、isAutoCollectをtrueへ

		if (Input::IsTrigger(KEY_INPUT_X) && m_stageManager->GetPlayer().GetBomb() > 0) {								// X押下且つ残ボムがある場合
			if (m_stageManager->GetPlayer().IsDeathTimerActive() || !m_stageManager->GetPlayer().IsInvincible()) {		//喰らいボム通常ボム発動した場合
				isAutoCollect = true;																				//回収モードオン
			}
		}

		m_bm->UpdateItems(
			m_stageManager->GetPlayer(),
			50.0f,
			isAutoCollect
		);

		bool isDead = m_stageManager->GetBoss().CheckCollision(m_stageManager->GetPlayer(), *m_bm);

		bool isPlayerHit = false;
		if (!m_stageManager->GetPlayer().IsInvincible()) {
			isPlayerHit = m_bm->CheckPlayerCollision(
				m_stageManager->GetPlayer().GetX(),
				m_stageManager->GetPlayer().GetY(),
				m_stageManager->GetPlayer().GetHitR()
			);
		}

		if (isPlayerHit && !m_stageManager->GetPlayer().IsDeathTimerActive()) {	//被弾且つdeathTimerがアクティブではない時
			m_stageManager->GetPlayer().StartDeathTimer();								//deathTimerを始動(アクティベート)
		}

		if (m_stageManager->GetPlayer().GetLife() < 0) {		//残機(整数型)が負即ち残機0で被弾した状態の場合
			GameOverMenu.SetSelect(0);							//ゲームオーバーシーンのカーソルの初期化
			scene = SCENE_GAMEOVER;
		}
		if (Input::IsTrigger(KEY_INPUT_X) && m_stageManager->GetPlayer().GetBomb() > 0) {// X押下且つボム数が1以上の時
			if (m_stageManager->GetPlayer().IsDeathTimerActive()) {								//deathTimerがアクティブな時
				m_stageManager->GetPlayer().StopDeathTimer();												//deathTimerを停止(inactivate)
				m_stageManager->GetPlayer().SetInVincible(m_stageManager->GetPlayer().GetInvincibleTime());	//無敵時間をセット
				m_bm->ConvertEnemyBulletsToItems();														//敵弾を小スコアへ変換
				m_stageManager->GetPlayer().ComsumeBomb();													//ボム消費
			}
			else if (!m_stageManager->GetPlayer().IsInvincible()) {									//無敵状態でなければ通常ボム消費	
				m_bm->ConvertEnemyBulletsToItems();														//敵弾を小スコアへ変換
				m_stageManager->GetPlayer().ComsumeBomb();													//ボム消費
			}
		}

		if (m_stageManager->IsStageClear()) {											//ボスを倒したフラグが返されたらタイトルに戻る
			while (Input::IsTrigger(KEY_INPUT_F)) ProcessMessage();		//z,enterキーを長押し状態でCLEARに遷移してきた時に離したらそのままタイトルに行くのを防ぐ為の文
			ClearInputCharBuf();				//画面遷移前にキー入力バッファを完全クリアして、入力画面でも予約されたキー入力が起きないように対策
			if (!playingPractice) {
				inputName = "";
				scene = SCENE_NAME_ENTRY;
			}
			else {
				scene = SCENE_CLEAR;
			}
		}

		if (Input::IsTrigger(KEY_INPUT_ESCAPE)) {
			PauseMenu.SetSelect(0);								//ポーズ遷移前にポーズメニュのカーソルを初期化
			scene = SCENE_PAUSE;
		}														//case SCENE_STAGEの中身を囲うことで変数のスコープがその中限定となる
	}
	break;



	case SCENE_GAMEOVER:
		if (Input::IsTrigger(KEY_INPUT_UP)) { GameOverMenu.Prev(); }
		if (Input::IsTrigger(KEY_INPUT_DOWN)) { GameOverMenu.Next(); }
		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			int select = GameOverMenu.GetSelect();

			if (select == 0) {					//コンティニューする場合
				if (playingBulletTest) {	//テストシーンの時
					testScene.GetPlayer().Reset(true);		//Practiceモード同様の残機残ボムのセット
					//testScene.GetPlayer().SetInVincible(testScene.GetPlayer().GetInvincibleTime());
					m_bm->ClearEnemyBullets();
					scene = SCENE_BULLET_TEST;
				}
				else {
					m_stageManager->GetPlayer().loadingLife();		//基準残機数のセット
					m_stageManager->GetPlayer().loadingBomb();		//基準ボム数のセット
					m_stageManager->GetPlayer().SetInVincible(m_stageManager->GetPlayer().GetInvincibleTime());
					scene = SCENE_STAGE;
					m_bm->ClearEnemyBullets();
					m_stageManager->GetPlayer().Reset(playingPractice);
				}
			}

			if (select == 1) {					//諦める場合
				if (playingBulletTest) {	//テストシーンの時
					playingBulletTest = false;
					scene = SCENE_TITLE;
				}
				else if (!playingPractice) {					//通常プレイの時のみ名前入力
					inputName = "";
					scene = SCENE_NAME_ENTRY;
				}
				else {
					scene = playingPractice ? SCENE_PRACTICE : SCENE_TITLE;


					m_scoreManager->AddScore("PLR", m_stageManager->GetPlayer().GetDisplayScore(), 1, 1, GetCurrentDateAsInt());
				}
				//scene = playingPractice ? SCENE_PRACTICE : SCENE_TITLE;
			}
		}
		break;



	case SCENE_CLEAR:
		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			scene = playingPractice ? SCENE_PRACTICE : SCENE_TITLE;													//三項演算子によりシーン遷移
		}
		break;


	case SCENE_PAUSE:
		if (Input::IsTrigger(KEY_INPUT_UP)) { PauseMenu.Prev(); }
		if (Input::IsTrigger(KEY_INPUT_DOWN)) { PauseMenu.Next(); }

		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			int select = PauseMenu.GetSelect();
			if (select == 0) scene = SCENE_STAGE;
			if (select == 1) { scene = playingPractice ? SCENE_PRACTICE : SCENE_TITLE; }
			if (select == 2) {
				m_bm->ClearAllBullets();
				m_bm->ClearAllItems();
				m_stageManager->Init(m_bm, lastSelectedStage, lastSelectedPhase, playingPractice);									//やり直しなのでリセット処理
				scene = SCENE_STAGE;
			}
		}

		if (Input::IsTrigger(KEY_INPUT_ESCAPE)) {
			scene = SCENE_STAGE;
		}

		if (Input::IsRelease(KEY_INPUT_R)) {
			m_bm->ClearAllBullets();
			m_bm->ClearAllItems();
			m_stageManager->Init(m_bm, lastSelectedStage, lastSelectedPhase, playingPractice);
			scene = SCENE_STAGE;
		}

		if (Input::IsTrigger(KEY_INPUT_Q)) {
			scene = playingPractice ? SCENE_PRACTICE : SCENE_TITLE;
		}
		break;



	case SCENE_RANKING:
		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN) || Input::IsTrigger(KEY_INPUT_ESCAPE)) {
			scene = SCENE_TITLE;
		}
		break;


	case SCENE_CORRUPTED_NOTICE:
		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			scene = SCENE_TITLE;
		}
		break;


	case SCENE_NAME_ENTRY:
	{
		char c = GetInputChar(TRUE);

		if (c != 0 && std::isalnum(static_cast<unsigned char>(c))) {	//アルファベットまたは数字の場合のみ追加 (std::isalnum を使用)
			char upperChar = static_cast<char>(std::toupper(c));

			if (inputName.length() < ScoreConfig::MAX_NAME_CHARS) {
				inputName += upperChar;									//文字数未満なら末尾に追加
			}
			else if (!inputName.empty()) {
				inputName.back() = upperChar;							//文字数上限に達している場合は末尾の文字を上書き
			}
		}

		if (Input::IsTrigger(KEY_INPUT_SPACE)) {
			if (inputName.length() < ScoreConfig::MAX_NAME_CHARS) {
				inputName += ' ';
			}
			else if (!inputName.empty()) {
				inputName.back() = ' ';									//スペースキーでも上書き対応
			}
		}
		if (Input::IsPress(KEY_INPUT_BACK)) {
			backspaceHoldCount++;
			if (backspaceHoldCount == 1 || (backspaceHoldCount > 15 && backspaceHoldCount % 4 == 0)) {	//最初の1フレーム、又は0.5秒長押し後に4フレームごとに連打
				if (!inputName.empty()) {
					inputName.pop_back();
				}
			}
		}
		else {
			backspaceHoldCount = 0;		//キーを離したら初期化
		}

		if (Input::IsTrigger(KEY_INPUT_RETURN)) {
			std::string finalName = inputName.empty() ? ScoreConfig::EMPTY_NAME_FALLBACK : inputName;
			int64_t finalScore = m_stageManager->GetPlayer().GetDisplayScore();
			uint32_t currentDate = GetCurrentDateAsInt();

			memset(&currentResultEntry, 0, sizeof(currentResultEntry));									//今回の結果情報を保持しておく
			strncpy_s(currentResultEntry.name, finalName.c_str(), ScoreConfig::MAX_NAME_BYTES - 1);
			currentResultEntry.score = finalScore;
			currentResultEntry.date = currentDate;

			m_scoreManager->AddScore(finalName, finalScore, 1, 1, currentDate);							//ランク外なら内部でソートせず弾く

			scene = SCENE_RANKING;
		}
	}
	break;



	case SCENE_QUIT_CONFIRM:
		if (Input::IsTrigger(KEY_INPUT_LEFT)) QuitMenu.Prev();
		if (Input::IsTrigger(KEY_INPUT_RIGHT)) QuitMenu.Next();

		if (Input::IsTrigger(KEY_INPUT_Z) || Input::IsTrigger(KEY_INPUT_RETURN)) {
			int select = QuitMenu.GetSelect();
			if (select == 0) scene = SCENE_TITLE;
			if (select == 1) DxLib::DxLib_End();
		}
		break;
	}
}

ScoreManager::ScoreManager(const std::string& filePath)		//ScoreManagerを生成する瞬間に自動で呼び出されるコンストラクタの開始でファイル名を受け取る
	: m_filePath(filePath) {		//受け取ったファイル名をクラス内部のm_filePathに保存

void ScoreManager::SetDefaultScores() {		
	std::memcpy(m_scoreData.header, ScoreConfig::FILE_HEADER.data(), ScoreConfig::FILE_HEADER.size());	//スコアデータの先頭識別子領域にファイルヘッダをセット
	m_scoreData.version = ScoreConfig::FILE_VERSION;		//スコアデータのバージョン番号スペースにバージョンを格納

	for (int i = 0; i < ScoreConfig::RANKING_MAX; ++i) {			//ランキングの枠数分ループ
		std::memset(m_scoreData.rank[i].name, 0, ScoreConfig::MAX_NAME_BYTES);			//名前格納領域をゼロクリア
		strncpy_s(m_scoreData.rank[i].name, ScoreConfig::DEFAULT_NAMES[i], ScoreConfig::MAX_NAME_BYTES - 1);	//初期ランキングネームセット
		m_scoreData.rank[i].score = ScoreConfig::DEFAULT_SCORES[i];				//初期スコアセット
		m_scoreData.rank[i].stage = ScoreConfig::DEFAULT_STAGE;					//初期到達ステージ番号セット
		m_scoreData.rank[i].difficulty = ScoreConfig::DEFAULT_DIFFICULTY;		//初期難易度番号セット
		m_scoreData.rank[i].date = ScoreConfig::DEFAULT_DATE;					//初期日付セット
	}
}


bool ScoreManager::Load() {
	std::ifstream ifs(m_filePath, std::ios::binary);		//ファイルをバイナリモードで開く準備
	if (!ifs.is_open()) {									//ファイルが開封できなかった(ファイル未作成)場合は初期スコアを新規ファイルとして書き出して作成して関数を途中終了
		Save();
		return false;
	}

	ScoreDataFile temp;			//ファイルから受け取ったデータを一時的に確認する為の作業用スペース
	ifs.read(reinterpret_cast<char*>(&temp), sizeof(ScoreDataFile));		//開いたファイルから、スコアデータ全体を一気に読み込みtempに入れる
	ifs.close();	//ファイルを閉じる

	if (std::memcmp(temp.header, ScoreConfig::FILE_HEADER.data(), ScoreConfig::FILE_HEADER.size()) != 0) {	//読み込んだヘッダが正しい識別子と不一致(データ破損か別のファイルである)場合
		SetDefaultScores();		//データ破損の為、メモリの中身を初期スコアに上書きして正常な状態に復旧して、初期化したことを通知して関数を途中終了
		Save();
		return false;
	}

	m_scoreData = temp;		//ヘッダが正常であるため、読み込んだデータを採用して関数を終了
	return true;
}

bool ScoreManager::Save() {		//メモリ上のスコアをファイルに書き出し保存する関数
	std::ofstream ofs(m_filePath, std::ios::binary);		//ファイルに書き込む為の準備をバイナリモードで開く
	if (!ofs.is_open()) {		//ファイルが開けなかった場合、保存に失敗したことを通知して関数を途中終了
		return false;
	}

	ofs.write(reinterpret_cast<const char*>(&m_scoreData), sizeof(ScoreDataFile));	//現在メモリにあるスコアデータ全体をファイルに丸ごと書き込む
	ofs.close();	//ファイルを閉じて、正常に保存できたことを通知して関数を終了
	return true;
}

bool ScoreManager::IsRankIn(int64_t score) const {	//今回のスコアがランクインしているかを確認する関数
	return score > m_scoreData.rank[ScoreConfig::RANKING_MAX - 1].score;	//今回のスコアが現在の一番低いスコアより大きいかを判定し、大きければtrueを返す
}

int ScoreManager::AddScore(const std::string& name, int64_t score, uint8_t stage, uint8_t difficulty, uint32_t date) {	//新規ランクインしたスコアを追加して並び替えて保存する関数
	if (!IsRankIn(score)) {		
		return -1;			//ランク外であれば関数終了
	}

	HighScoreEntry& last = m_scoreData.rank[ScoreConfig::RANKING_MAX - 1];	//一番低いデータを直接書き換えるので、その場所を示す
	std::memset(last.name, 0, ScoreConfig::MAX_NAME_BYTES);					//名前の記憶領域をゼロクリア
	strncpy_s(last.name, name.c_str(), ScoreConfig::MAX_NAME_BYTES - 1);	//今回の新規ネームを書き込む
	last.score = score;														//↓取得スコア、到達ステージ、難易度、達成年月日書き込み
	last.stage = stage;														
	last.difficulty = difficulty;											
	last.date = date;														

	std::sort(m_scoreData.rank, m_scoreData.rank + ScoreConfig::RANKING_MAX,	//ランキングデータを降順でソートする関数
		[](const HighScoreEntry& a, const HighScoreEntry& b) {		//a,bを比較
			return a.score > b.score;		//スコアが大きい方を前へ
		});

	int rankedIndex = -1;		//今回のデータが何位かを格納
	for (int i = 0; i < ScoreConfig::RANKING_MAX; ++i) {	//ランキングの大きさだけ繰り返す
		if (m_scoreData.rank[i].score == score && std::strcmp(m_scoreData.rank[i].name, name.c_str()) == 0) {	//i番目のスコアが今回のスコアと等しく且つ、名前も一致していた場合
			rankedIndex = i;	//その場所の番号を格納してループ脱却
			break;
		}
	}

	Save();					//更新されたランキングをファイルへ上書き保存して、順位を返して関数終了
	return rankedIndex;
}

const HighScoreEntry& ScoreManager::GetEntry(int rankIndex) const {		//画面に描画する為に、指定した順位のデータを取り出す関数
	if (rankIndex < 0 || rankIndex >= ScoreConfig::RANKING_MAX) {
		return m_scoreData.rank[0];		//指定された順位番号が負、又はランキング枠の大きさを超過している場合、エラーを防止する代わりに1位(Indexが0)を返す
	}
	return m_scoreData.rank[rankIndex];		//正常な番号であれば指定された順位データを返す
}

/*
				--- RotateAllの回転速度(rotSpeed)算出メモ ---
1回の発射で進む角度をΔθ[deg]としたとき、
全方位弾の隙間(step=360/way)に対してΔθが整数倍に近いとほぼ重なって回転してしまう。
前の弾の隙間の真ん中に配置して最もバラけさせるには、
	Δθ=step*(k+1/2) (k=0,1,2⋯)
となるようにrotSpeedを定めれば良い。
	rotSpeed=(Δθ/(si*180/π))
			=(Δθ*π)/(si*180)
綺麗にバラけるΔθの選び方は、、
	step=360.0f/way
であり、Δθ=step*0.5f(半周期ズレ)やstep*1.5f, step*2.5f等がある。

計算例
(si=20,way=36で(1/3 + 2)周期ズレの場合)
・step=360.0f/36=10.0f
1/3周期ズレより1/3 step=10/3≒3.33で、回転感を出すために適当なk=2周期を加えると、Δθ=(1/3 +2)step≒10*2.33=23.3f
	rotSpeed=(23.3f*3.14f)/(20*180)=0.0203227778≒0.0203f

(si=10,way=12で(1/3 + 1)周期ズレの場合)
・step=360.0f/12=30.0f
・Δθ=30.0f*(1/3 +1)=30.0f(4/3)=40.0f
	rotSpeed=(40.0f*3.14f)/(10*180)=0.069033333≒0.0690f
*/

void StageManager::Update(BulletManager* bm) {
	stageTimer++;								//時間を進める

	if (state == StageState::DOCHU) {
		if (stageTimer == 700) {
			state = StageState::MID_BOSS_BATTLE;						//道中ボス
			float X = Left + (Right - Left) / 2.0f;
			float Y = Top - (Bottom - Top)  / 4.0f;
			boss.Spawn(X, Y, Stage1Data::MidBossPhases, Stage1Data::MID_BOSS_PHASE_MAX, 0, -1);										
		}
		
		for (int i = 0; i < Stage1Data::SPAWN_COUNT; i++) {
			if (Stage1Data::Timeline[i].frame == stageTimer) {				//EnemySpawnのframeと経過時間が一致した場合
				for (int s = 0; s < MAX_SPAWN_SLOT; s++) {
					if (!SpawnSlots[s].IsActive()) {
						SpawnSlots[s].Activate(Stage1Data::Timeline[i].count, i);		//引数でiを渡してどれを使うか覚えさせる
						break;
					}
				}
			}
		}
		for (int s = 0; s < MAX_SPAWN_SLOT; s++) {
			if (SpawnSlots[s].IsActive()) {
				int dataIdx = SpawnSlots[s].GetDataIdx();

				if (SpawnSlots[s].Tick(Stage1Data::Timeline[dataIdx].interval)) {
					for (int j = 0; j < Enemy::EnemyMax; j++) {
						if (!fairies[j].GetFlag()) {
							auto& d = Stage1Data::Timeline[dataIdx];
							fairies[j].Encount(d.x, d.y, d.r, d.vx, d.vy, d.hp,d.shotConf,d.clearOnEnd, d.dropCounts);
							break;
						}
					}
				}
			}
		}


		if (stageTimer >= 4000) {
			state = StageState::BOSS_BATTLE;		//ステージ状態をボス戦へ移行
			float X = Left + (Right - Left) / 2.0f;
			float Y = Top + (Bottom - Top) / 4.0f;
			boss.Spawn(X, Y, Stage1Data::BossPhases, Stage1Data::BOSS_PHASE_MAX, 0, -1);				//ボスはフェーズ1から最終フェーズまで
		}
	}
	else if (state == StageState::MID_BOSS_BATTLE) {
		boss.Update(player, bm);
		if (!boss.GetFlag()) {						//ボスフラグが消えたら道中復帰
			boss.Reset();							//中ボス終了後にボス状態をリセット
			state = StageState::DOCHU;
		}
	}

	player.Update(bm);								//ヘッダでもらったbmを渡して自機が弾を撃てるようにする

	for (int i = 0; i < Enemy::EnemyMax; i++) {
		if (fairies[i].GetFlag()) {
			fairies[i].Update(player, bm);					//フラグが立っている妖精を見つけたら更新して動かす

		}
	}

	if (state == StageState::BOSS_BATTLE) {
		boss.Update(player, bm);							//ボス戦中の場合、bmでボスの更新をしてボスが弾を撃てるようにする
	}
}

void StageManager::Draw(BulletManager* bm) {		//弾とかの描画
	player.Draw();									//自機の描画


	for (int i = 0; i < Enemy::EnemyMax; i++) {
		if (fairies[i].GetFlag()) {
			if (fairies[i].CheckCollision(player, bm)) {
				if (fairies[i].IsClearOnDeath()) {
					bm->ConvertEnemyBulletsToItems();
				}
			}
			fairies[i].Draw();						//フラグが立っている妖精を描画
		}
	}

	if (state==StageState::MID_BOSS_BATTLE || state == StageState::BOSS_BATTLE) {
		boss.Draw();								//道中ボス・ボス戦中のボスの描画
	}
}

void StageManager::Init(BulletManager* bm, int stageNum, int startPhaseIdx, bool isPractice) {							//ステージの状態を全てリセットする関数
	currentStageNum = stageNum;
	
	bm->ConvertEnemyBulletsToItems();
	bm->ClearAllItems();

	player.Reset(isPractice);
	boss.Reset();									//ボスフラグを折る


	for (int i = 0; i < Enemy::EnemyMax;i++) {
		fairies[i].SetFlag(false);					//妖精フラグも折る
	}

	for (int i = 0; i < MAX_SPAWN_SLOT; i++) {
		SpawnSlots[i].Reset();
	}
	if (startPhaseIdx > 0) {							//Practiceモードの時はボス戦までスキップ
		stageTimer = 2000;							//ステージタイマーを1200(通常時のボスが出てくる時間)にすることで道中の雑魚が出現するのを防止
		state = StageState::BOSS_BATTLE;			//状態をボス戦へ

		float X = Left + (Right - Left) / 2.0f;
		float Y = Top + (Bottom - Top) / 4.0f;
		boss.Spawn(X, Y, Stage1Data::BossPhases, Stage1Data::BOSS_PHASE_MAX, startPhaseIdx - 1, - 1);				//画面中央にボスを出現させ、指定フェーズへジャンプ
	}
	else {
		stageTimer = 0;
		state = StageState::DOCHU;
	}
}

#pragma once
#include "Boss.h"
#include "Enemy.h"
#include "Player.h"
#include "BulletManager.h"
#include "BurstController.h"
#include "BulletPattern.h"


enum class StageState {
	ST_TITLE,					//道中のタイトル表示
	DOCHU,						//道中
	MID_BOSS_BATTLE,			//道中ボス
	BOSS_APPEAR,				//ボス出現(会話等)
	BOSS_BATTLE,				//ボス戦
	BOSS_END,					//ボス討伐完了演出
	CLEAR						//クリア演出
};

struct EnemySpawn {
	int   frame;							//開始から何フレーム目に出すか
	float x, y;								//出現位置
	float vx, vy;							//移動速度
	float r;								//半径
	int	  hp;								//体力
	int   count;							//隊列の数				
	int   interval;							//出撃間隔(フレーム)
	BulletPattern::ShotConfig shotConf;		//弾のパラメータ
	bool clearOnEnd;						//弾消しフラグ
	int dropCounts[6];						//妖精が死んだ時に落とすアイテムの種類(0：小スコア, 1：スコア, 2：パワー, 3：残機, 4：ボム, 5：大パワー)
};

class StageManager {
private:
	StageState state = StageState::DOCHU;		//道中のインスタンス作成
	int stageTimer = 0;							//フレームカウンタ

	Player player;								//自機インスタンス
	Boss boss;									//ボスインスタンス
	Enemy fairies[100];							//妖精配列インスタンス
	int currentStageNum = 1;					//現在何面をプレイしているか

public:
	static constexpr int upMargin = 80;			//妖精が出てくる時に上からどれくらい開けて出現するか
	static constexpr int MAX_SPAWN_SLOT = 10;	//同時出現可能隊数
	BurstController SpawnSlots[MAX_SPAWN_SLOT];		//出現予約スロット

	StageManager() : stageTimer(0),state(StageState::DOCHU){}	//コンストラクタというクラスの初期化専用関数で、クラスと同じ名前にして、戻り値の型を記述しない。クラスが生まれた瞬間stageTimerを0に、stateをDOCHUに設定する役割

	void Update(BulletManager* bm);			//弾を出すタイミングとかで必要になるので住所を知っておく。(&bmは場所を教える。*bmはその場所の情報を受け取る箱のようなもの)
	void Draw(BulletManager* bm);
	void Init(BulletManager* bm, int stageNum = 1, int startPhaseIdx = 0, bool isPractice = false);			//ステージ数と開始段階インデックス(0:最初から, 1:中ボス, 2:ボス通常)を組み合わせる設計

	void ReloadGraphic() {
		boss.ReloadGraphic();					
	}

	Player& GetPlayer() { return player; }		//&は参照渡しで、privateに隔離されてるインスタンスそのものへのアクセス権を引き出している。窓口の様なもの
	Boss& GetBoss() { return boss; }
	Enemy* GetFairies() { return fairies; }

	bool IsStageClear() const { return (state == StageState::BOSS_BATTLE && boss.IsDead()); }		//ボス戦中でボスの全フェーズが終了した時にステージクリアと判定
};
